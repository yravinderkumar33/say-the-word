// The debug status exposes trayHandsFree, not the actual native menu. This probe
// therefore combines a real processing state with the actual AppTray template,
// loaded under counting Electron stubs; it does not open the system menu.
scenario('QA5: processing Cancel on the pill must also be offered by the tray menu', async () => {
  const requireQa5 = createRequire(import.meta.url)
  const tsQa5 = requireQa5('typescript')
  const vmQa5 = requireQa5('node:vm')
  const traySourceQa5 = readFileSync(join(root, 'src/main/windows/tray.ts'), 'utf8')
  let latestQa5 = []
  const electronQa5 = {
    Menu: { buildFromTemplate: (items) => { latestQa5 = items; return items } },
    Tray: class { setToolTip() {} on() {} setContextMenu() {} setImage() {} },
    app: { getName: () => 'QA App', quit() {} },
  }
  const exportsQa5 = {}
  const sourceQa5 = tsQa5.transpileModule(traySourceQa5, {
    compilerOptions: { module: tsQa5.ModuleKind.CommonJS, target: tsQa5.ScriptTarget.ES2022 },
  }).outputText
  vmQa5.runInNewContext(sourceQa5, {
    exports: exportsQa5,
    require: (id) => {
      if (id === 'electron') return electronQa5
      if (id === '@shared/icon-shapes') return { trayIconState: () => 'ready' }
      if (id === '@shared/microphone-name') return { microphoneName: (s) => s, shortenInTheMiddle: (s) => s }
      if (id === '../dictation/pause') return { clockTime: () => '12:00' }
      if (id === './tray-icon') return { trayImage: () => null }
      throw new Error(`Unexpected safe probe dependency ${id}`)
    },
  })
  const stubTrayQa5 = new exportsQa5.AppTray(new Proxy({}, { get: () => () => {} }))
  send('mode verbatim')
  send('text-delay 10000')
  try {
    play(SHORT)
    await press()
    await sleep(SHORT.ms + 300)
    send('release')
    const processingQa5 = await waitUntil('processing pill Cancel', (now) => now.pill === 'processing' && now.pillCancel === true, 6_000)
    stubTrayQa5.update({ handsFree: processingQa5.trayHandsFree, offer: processingQa5.trayOffer })
    const cancelItemsQa5 = latestQa5.filter((item) => item.label === 'Cancel Dictation').length
    expect(cancelItemsQa5 > 0, `pill Cancel=true; tray Cancel items=${cancelItemsQa5}; native menu template evaluated with the live processing state`)
  } finally {
    send('escape')
    send('text-delay 0')
    await waitIdle(20_000)
  }
})

scenario('QA5: latest Cleanup model choice wins while earlier validation is slow', async () => {
  await inCleanedMode('echo', async () => {
    await showPage('home', '')
    await showPage('cleanup', 'Cleanup')
    // Allow its initial choices to be listed before delaying subsequent validation.
    await sleep(3_000)
    ollama.tagsDelayMs = 900
    try {
      let fromQa5 = log.length
      send('choose-hub Model leaky:latest')
      await waitForLog('hub field "Model": set', 3_000, fromQa5)
      await sleep(150)
      fromQa5 = log.length
      // type-hub decodes an omitted value as empty, which is the Rules only option.
      send('type-hub Model')
      await waitForLog('hub field "Model": set', 3_000, fromQa5)
      await waitUntil('last choice Rules only applied', (now) => now.prefs.cleanupModel === null, 2_000)
      await sleep(3_500)
      const afterQa5 = await status()
      expect(afterQa5.prefs.cleanupModel === null, 'the earlier model selection overwrote the later Rules only choice after validation completed')
    } finally {
      ollama.tagsDelayMs = 0
    }
  })
})
